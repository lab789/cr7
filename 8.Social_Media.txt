╔══════════════════════════════════════╗
║      STEP 6 : AGGREGATION           ║
╚══════════════════════════════════════╝

db.posts.aggregate([
{
    $group: {
        _id: null,
        TotalLikes: { $sum: "$likes" }
    }
}
])

db.posts.aggregate([
{
    $group: {
        _id: null,
        AverageLikes: { $avg: "$likes" }
    }
}
])

db.posts.aggregate([
{
    $unwind: "$hashtags"
},
{
    $group: {
        _id: "$hashtags",
        Count: { $sum: 1 }
    }
},
{
    $sort: {
        Count: -1
    }
}
])

db.posts.aggregate([
{
    $sort: {
        likes: -1
    }
},
{
    $limit: 1
}
])

db.posts.aggregate([
{
    $group: {
        _id: "$location",
        TotalPosts: { $sum: 1 }
    }
}
])
